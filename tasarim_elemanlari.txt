    ana_pencere = parent.winfo_toplevel()
    ANA_YÜKLEME_GÖSTERGESI = ctk.CTkFrame(ana_pencere, fg_color=["#EAEAEA", "#1E1E1E"], corner_radius=10, border_width=2, border_color="#1F6AA5")
    
    lbl_ana_load = ctk.CTkLabel(ANA_YÜKLEME_GÖSTERGESI, text="⚡ Görsel Sunucuya Yükleniyor...", font=ctk.CTkFont(size=13, weight="bold"))
    lbl_ana_load.pack(padx=40, pady=(20, 10))

    ANALOG_PROGRESS_BAR.pack(padx=40, pady=(0, 20))

    txt_output._textbox.tag_configure("center", justify="center")

    txt_output._textbox.tag_configure("serit_resim", foreground="#FFD700", font=ctk.CTkFont(family="Courier", size=13, weight="bold"))
    txt_output._textbox.tag_configure("dil_resim", foreground="#107C41")
    txt_output._textbox.tag_configure("bbcode_tag", foreground="#00BFFF", font=ctk.CTkFont(family="Courier", size=13, weight="bold"))
    txt_output._textbox.tag_configure("bbcode_url", foreground="#FF8C00", font=ctk.CTkFont(family="Courier", size=13))
    def bbcode_renklendir_tetikleyici(event=None):
        content = txt_output.get("1.0", "end-1c")
        txt_output._textbox.tag_remove("bbcode_tag", "1.0", "end")
        txt_output._textbox.tag_remove("bbcode_url", "1.0", "end")
        txt_output._textbox.tag_remove("serit_resim", "1.0", "end")
        
        for match in re.finditer(r'\[\/?[a-zA-Z0-9\=\#\s\:\/\.\-\_\?\&\=\%\+\,\;\!\@\^\*\(\)]+\]', content):
            txt_output._textbox.tag_add("bbcode_tag", f"1.0 + {match.start()} chars", f"1.0 + {match.end()} chars")
            
        for match in re.finditer(r'https?://[^\s\[\]]+', content):
            txt_output._textbox.tag_add("bbcode_url", f"1.0 + {match.start()} chars", f"1.0 + {match.end()} chars")

        for match in re.finditer(r'\[img\]https?://[^\s\[\]]+\[/img\]', content, re.IGNORECASE):
            txt_output._textbox.tag_add("serit_resim", f"1.0 + {match.start()} chars", f"1.0 + {match.end()} chars")

    def sarmalanmis_insert(index, text, *args):
        txt_output._textbox._original_insert(index, text, *args)
        bbcode_renklendir_tetikleyici()
    txt_output._textbox._original_insert = txt_output._textbox.insert
    txt_output._textbox.insert = sarmalanmis_insert
    txt_output.bind("<KeyRelease>", bbcode_renklendir_tetikleyici)
    txt_output.bbcode_renklendir = bbcode_renklendir_tetikleyici
    
    return sol_frame, txt_output, sag_frame

def tam_otomatik_imgbb_yukle_worker(girdi_kutusu, update_callback, dosya_yolu):
    """Playwright otomasyon mekanizması üzerinden saf kaynak ayıklayan arka plan işçisi."""
    try:
        with sync_playwright() as p:
            browser = p.chromium.launch(headless=True)
            context = browser.new_context(viewport={"width": 1280, "height": 720})
            page = context.new_page()
            page.route("**/*", lambda route: route.abort() if route.request.resource_type in ["image", "stylesheet", "font", "media"] else route.continue_())
            page.goto("https://imgbb.com", wait_until="domcontentloaded")
            page.wait_for_selector('li[data-action="top-bar-upload"]', timeout=10000)
            page.click('li[data-action="top-bar-upload"]')
            page.set_input_files('input[type="file"]', dosya_yolu)
            page.wait_for_selector('button[data-action="upload"]', state="visible", timeout=10000)
            page.click('button[data-action="upload"]')
            page.wait_for_selector('select[id="uploaded-embed-toggle"]', state="visible", timeout=20000)
            page.select_option('select[id="uploaded-embed-toggle"]', value="viewer-links")
            link = page.locator('#uploaded-embed-code-0').input_value()
            browser.close()
            
            # 🚨 DÜZELTME: KUTUNUN İÇİNE ARTIK [img] DEĞİL SADECE SAF WEB LİNKİ BASILARAK KISIRDÖNGÜ KIRILDI 🚨
            if link and "ibb.co" in link:
                def arayüze_yaz():
                    if isinstance(girdi_kutusu, ctk.CTkTextbox):
                        girdi_kutusu.delete("1.0", "end")
                        girdi_kutusu.insert("1.0", link.strip())
                    else:
                        girdi_kutusu.delete(0, 'end')
                        girdi_kutusu.insert(0, link.strip())
                    if hasattr(girdi_kutusu, "renk_tazeleyici"): girdi_kutusu.renk_tazeleyici()
                    if update_callback: update_callback()
                if ANA_YÜKLEME_GÖSTERGESI:
                    ANA_YÜKLEME_GÖSTERGESI.after(10, arayüze_yaz)
    except Exception as e:
        if ANA_YÜKLEME_GÖSTERGESI:
            ANA_YÜKLEME_GÖSTERGESI.after(10, lambda: messagebox.showerror("Hata", f"Otomatik yükleme motorunda hata oluştu:\n{str(e)}"))
    finally:
        if ANA_YÜKLEME_GÖSTERGESI:
            def paneli_kapat():
                if ANALOG_PROGRESS_BAR: ANALOG_PROGRESS_BAR.stop()
                ANA_YÜKLEME_GÖSTERGESI.place_forget()
            ANA_YÜKLEME_GÖSTERGESI.after(10, paneli_kapat)

    def asenkron_gorsel_yukleyici(self, url, index_pos):
        """Kilitlenmeleri çözen ve link dönüşümlerini mühürleyen asenkron resim indirme motoru."""
        try:
            ham_url = url.strip()
            if "ibb.co/" in ham_url and "i.ibb.co" not in ham_url:
                res = requests.get(ham_url, timeout=4, headers={"User-Agent": "Mozilla/5.0"})
                if res.status_code == 200:
                    m = re.findall(r'property="og:image" content="(https://i\.ibb\.co/[^\s"<>]+)"', res.text)
                    if m: ham_url = m[0]
                    else:
                        m_alt = re.findall(r'src="(https://i\.ibb\.co/[^\s"<>]+)"', res.text)
                        if m_alt: ham_url = m_alt[0]

            if not ham_url.startswith("http"): return
            if ham_url in self.resim_hafizasi:
                tk_img = self.resim_hafizasi[ham_url]
                self.after(5, lambda: self.arayuze_gorsel_enjekte(index_pos, tk_img))
                return
            cevap = requests.get(ham_url, timeout=5, headers={"User-Agent": "Mozilla/5.0"})
            if cevap.status_code == 200:
                img = Image.open(io.BytesIO(cevap.content))
                genislik = 460 if "images/" in ham_url or "i.ibb.co" in ham_url or "imgur.com" in ham_url else 220
                
                # 🚨 KESİN DÜZELTME: TUPLE HATA MATRİSİNİ SIFIRLAYAN PİKSEL BOYUTLANDIRMA İNDEKSLERİ MÜHÜRLENDİ 🚨
                oran = genislik / float(img.size[0])
                yukseklik = int(float(img.size[1]) * oran)
                img = img.resize((genislik, yukseklik), Image.Resampling.LANCZOS)
                
                tk_img = ImageTk.PhotoImage(img)
                self.resim_hafizasi[ham_url] = tk_img
                self.after(5, lambda: self.arayuze_gorsel_enjekte(index_pos, tk_img))
        except Exception: pass

    def asenkron_youtube_kart_isleyici(self, url, index_pos):
        try:
            v_id = "dQw4w9WgXcQ"
            m = re.search(r'(?:v=|\/v\/|embed\/|youtu\.be\/|\/shorts\/|watch\?v=)([a-zA-Z0-9_-]{11})', url)
            if m: v_id = m.group(1)
            kapak = f"https://youtube.com{v_id}/hqdefault.jpg"
            if kapak in self.resim_hafizasi:
                tk_vid = self.resim_hafizasi[kapak]
                self.after(5, lambda: self.arayuze_gorsel_enjekte(index_pos, tk_vid))
                return

            res = requests.get(kapak, timeout=5, headers={"User-Agent": "Mozilla/5.0"})
            if res.status_code == 200:
                img = Image.open(io.BytesIO(res.content)).convert("RGBA")
                img = img.resize((350, 220), Image.Resampling.LANCZOS).crop((0, 20, 350, 200))
                ov = Image.new("RGBA", img.size, (0, 0, 0, 0))
                dr = ImageDraw.Draw(ov)
                dr.rounded_rectangle([0, 0, img.size[0]-1, img.size[1]-1], radius=12, fill=(228, 20, 24, 220))
                dr.polygon([(160, 85), (160, 115), (180, 100)], fill=(255, 255, 255, 255))
                dr.rectangle([0, 0, img.size[0]-1, img.size[1]-1], outline=(120, 120, 120, 255), width=1)
                final = Image.alpha_composite(img, ov)
                
                tk_vid = ImageTk.PhotoImage(final)
                self.resim_hafizasi[kapak] = tk_vid
                self.after(5, lambda: self.arayuze_gorsel_enjekte(index_pos, tk_vid))
        except Exception: pass

    def arayuze_gorsel_enjekte(self, index_pos, tk_img_obj):
        try:
            self.txt_view.configure(state="normal")
            self.txt_view._textbox.image_create(index_pos, image=tk_img_obj)
            self.txt_view.configure(state="disabled")
            self.txt_view._textbox.update_idletasks()
        except Exception: pass
    def bbcode_grafik_parser_tetikleyici(self, text):
        """%100 SAF STANDART ETİKET TARAYAN BBCODE FORUM UYUMLU MOTOR"""
        temiz_veri = text.replace("\r\n", "\n").strip()
        lines = temiz_veri.split("\n")
        ilk_satir = False
        
        for line in lines:
            line_str = line.strip()
            if not line_str or line_str.lower() in ["center", "img]", "/img]", "/center]", "[center]", "[/center]"]: 
                continue

            if not ilk_satir and "[img]" not in line_str.lower() and "[center]" not in line_str.lower() and "[code]" not in line_str.lower():
                self.after(5, lambda s=line_str: (self.txt_view.configure(state="normal"), self.txt_view.insert("end", s + "\n\n", ("p_center", "sunum_adi_stili")), self.txt_view.configure(state="disabled")))
                ilk_satir = True
                continue

            if "[code]" in line_str.lower() or "[/code]" in line_str.lower():
                cl = re.sub(r'\[\/?code\]|\[\/?center\]', '', line_str, flags=re.IGNORECASE).strip()
                if cl:
                    self.after(5, lambda c=cl: (self.txt_view.configure(state="normal"), self.txt_view.insert("end", "\n\n  Code:\n  " + c + "  \n\n", "code_stili"), self.txt_view.configure(state="disabled")))
                continue
            if "[img]" in line_str.lower():
                links = re.findall(r'\[img\](https?://[^\s\[\]]+)\[/img\]', line_str, re.IGNORECASE)
                if links:
                    for url in links:
                        idx = self.txt_view.index("end-1c")
                        self.after(5, lambda i=idx: (self.txt_view.configure(state="normal"), self.txt_view.insert("end", "\n", "p_center"), self.txt_view.configure(state="disabled")))
                        RESİM_HAVUZU.submit(self.asenkron_gorsel_yukleyici, url, idx)
                    self.after(5, lambda: (self.txt_view.configure(state="normal"), self.txt_view.insert("end", "\n"), self.txt_view.configure(state="disabled")))
                continue

            if "[youtube]" in line_str.lower():
                yt = re.search(r'\[youtube\](https?://[^\s\[\]]+)\[/youtube\]', line_str, re.IGNORECASE)
                if yt:
                    yt_url = yt.group(1).strip()
                    idx_s = self.txt_view.index("end-1c")
                    self.after(5, lambda i=idx_s: (self.txt_view.configure(state="normal"), self.txt_view.insert("end", "\n", "p_center"), self.txt_view.configure(state="disabled")))
                    RESİM_HAVUZU.submit(self.asenkron_gorsel_yukleyici, "https://bitturk.net", idx_s)
                    
                    idx_v = self.txt_view.index("end-1c")
                    self.after(5, lambda i=idx_v: (self.txt_view.configure(state="normal"), self.txt_view.insert("end", "\n\n", "p_center"), self.txt_view.configure(state="disabled")))
                    RESİM_HAVUZU.submit(self.asenkron_youtube_kart_isleyici, yt_url, idx_v)
                    self.after(5, lambda: (self.txt_view.configure(state="normal"), self.txt_view.insert("end", "\n"), self.txt_view.configure(state="disabled")))
                continue

            cl = re.sub(r'\[\/?center\]|\[\/?b\]', '', line_str, flags=re.IGNORECASE)
            tags = ["p_bold"]
            if "[center]" in line_str.lower(): tags.append("p_center")
            if cl.strip(): 
                self.after(5, lambda c=cl, t=tags: (self.txt_view.configure(state="normal"), self.txt_view.insert("end", c + "\n", tuple(t)), self.txt_view.configure(state="disabled")))

def ekle_girdi(parent, label_text, is_textbox=False, height=60, has_upload=False, ornek="", update_callback=None):
    tekli_container = ctk.CTkFrame(parent, border_width=1, border_color=["#E5E7EB", "#374151"], fg_color="transparent")
    tekli_container.pack(fill="x", pady=3, padx=2)

    lbl.pack(fill="x", pady=(3, 1), padx=4)
    varsayilan_fg = ["#F9F9FA", "#343638"] if ctk.get_appearance_mode() == "Dark" else ["#F9F9FA", "#F9F9FA"]

    def girdi_renk_guncelle(girdi_obj):
